def get_sentinel2_collection(geometry, start_date, end_date):
    """Get Sentinel-2 collection with cloud filtering."""
    collection = ee.ImageCollection('COPERNICUS/S2_SR_HARMONIZED') \
        .filterBounds(geometry) \
        .filterDate(start_date, end_date) \
        .filter(ee.Filter.lt('CLOUDY_PIXEL_PERCENTAGE', CONFIG['sentinel2']['cloud_threshold']))
    
    return collection

def calculate_indices(image):
    """Calculate NDVI, SAVI, and LAI for an image."""
    # Scale to 0-1
    image = image.divide(10000)
    
    # Get bands
    nir = image.select('B8')
    red = image.select('B4')
    
    # NDVI
    ndvi = nir.subtract(red).divide(nir.add(red)).rename('NDVI')
    
    # SAVI (L=0.5)
    savi = nir.subtract(red).multiply(1.5).divide(nir.add(red).add(0.5)).rename('SAVI')
    
    # LAI (simplified empirical)
    lai = ndvi.multiply(3.5).add(0.2).rename('LAI')
    lai = lai.where(lai.lt(0), 0).where(lai.gt(8), 8)
    
    # Combine
    result = ee.Image.cat([ndvi, savi, lai])
    return result.set('system:time_start', image.get('system:time_start'))

def download_image_data_geemap(image, geometry, date_str):
    """Download image data using geemap (more reliable)."""
    try:
        # Create a temporary directory
        temp_dir = tempfile.mkdtemp()
        filename = f"sentinel2_{date_str}.tif"
        filepath = os.path.join(temp_dir, filename)
        
        # Download using geemap
        geemap.ee_export_image(
            image,
            filename=filepath,
            scale=CONFIG['sentinel2']['scale'],
            region=geometry,
            file_per_band=False
        )
        
        # Check if file was created
        if os.path.exists(filepath):
            print(f"  ✅ Downloaded: {filepath}")
            
            # Try to read with rasterio
            try:
                import rasterio
                with rasterio.open(filepath) as src:
                    data = src.read()  # Shape: (bands, height, width)
                    
                # Reshape to (pixels, bands)
                bands, height, width = data.shape
                data_reshaped = data.transpose(1, 2, 0).reshape(-1, bands)
                
                # Cleanup
                os.remove(filepath)
                os.rmdir(temp_dir)
                
                return data_reshaped
                
            except Exception as e:
                print(f"  ❌ Error reading file: {e}")
                return None
        else:
            print(f"  ❌ File not created: {filepath}")
            return None
            
    except Exception as e:
        print(f"  ❌ Download error: {e}")
        return None

def download_image_data_simple(image, geometry, date_str):
    """Simpler download using sampleRectangle."""
    try:
        # Get data as dictionary
        data_dict = image.sampleRectangle(
            region=geometry,
            defaultValue=0
        ).getInfo()
        
        # Extract the bands
        ndvi_data = np.array(data_dict['properties']['NDVI'])
        savi_data = np.array(data_dict['properties']['SAVI'])
        lai_data = np.array(data_dict['properties']['LAI'])
        
        # Stack into (height, width, bands)
        stacked = np.stack([ndvi_data, savi_data, lai_data], axis=-1)
        
        # Reshape to (pixels, bands)
        height, width, bands = stacked.shape
        data_reshaped = stacked.reshape(-1, bands)
        
        return data_reshaped
        
    except Exception as e:
        print(f"  ❌ Simple download failed: {e}")
        
        # Try alternative method
        try:
            print(f"  Trying alternative method...")
            
            # Get the image as array using reduceRegion
            stats = image.reduceRegion(
                reducer=ee.Reducer.toList(),
                geometry=geometry,
                scale=CONFIG['sentinel2']['scale'],
                maxPixels=1e6
            ).getInfo()
            
            # Extract arrays
            ndvi_vals = stats['NDVI']
            savi_vals = stats['SAVI']
            lai_vals = stats['LAI']
            
            if ndvi_vals and len(ndvi_vals) > 0:
                data = np.column_stack([ndvi_vals, savi_vals, lai_vals])
                print(f"  ✅ Got {len(data)} pixels via reduceRegion")
                return data
            else:
                return None
                
        except Exception as e2:
            print(f"  ❌ Alternative method also failed: {e2}")
            return None

def mask_s2_clouds(image):
    qa = image.select('QA60')
    cloud_bitmask = 1 << 10
    cirrus_bitmask = 1 << 11
    mask = qa.bitwiseAnd(cloud_bitmask).eq(0).And(qa.bitwiseAnd(cirrus_bitmask).eq(0))
    return image.divide(10000).copyProperties(image, ['system:time_start'])

def calculate_indices(image):
    nir = image.select('B8')
    red = image.select('B4')
    ndvi = nir.subtract(red).divide(nir.add(red)).rename('NDVI')
    savi = nir.subtract(red).multiply(1.5).divide(nir.add(red).add(0.5)).rename('SAVI')
    lai = ndvi.multiply(3.5).add(0.2).rename('LAI')
    lai = lai.where(lai.lt(0), 0).where(lai.gt(8), 8)
    return image.addBands([ndvi, savi, lai])

def get_sentinel2_image(geometry, target_date, days_buffer=7):
    start = ee.Date(target_date).advance(-days_buffer, 'day')
    end = ee.Date(target_date).advance(days_buffer, 'day')
    
    collection = ee.ImageCollection('COPERNICUS/S2_SR_HARMONIZED') \
        .filterBounds(geometry) \
        .filterDate(start, end) \
        .filter(ee.Filter.lt('CLOUDY_PIXEL_PERCENTAGE', 20)) \
        .map(mask_s2_clouds) \
        .map(calculate_indices)
    
    if collection.size().getInfo() == 0:
        return None, None
    
    image = ee.Image(collection.first())
    actual_date = ee.Date(image.get('system:time_start')).format('YYYY-MM-dd').getInfo()
    
    return image, actual_date

def image_to_array(image, geometry, scale):
    """Simplified array extraction using sampleRectangle."""
    bbox = geometry.bounds().getInfo()['coordinates'][0]
    lons = [p[0] for p in bbox]
    lats = [p[1] for p in bbox]
    bounds = [min(lons), min(lats), max(lons), max(lats)]
    
    indices_image = image.select(['NDVI', 'SAVI', 'LAI'])
    sampled = indices_image.sampleRectangle(region=geometry.bounds(), defaultValue=0)
    
    ndvi = np.array(sampled.get('NDVI').getInfo())
    savi = np.array(sampled.get('SAVI').getInfo())
    lai = np.array(sampled.get('LAI').getInfo())
    
    height, width = ndvi.shape
    transform = from_bounds(bounds[0], bounds[1], bounds[2], bounds[3], width, height)
    crs = 'EPSG:4326'
    
    return ndvi, savi, lai, transform, crs

def apply_pca(ndvi, savi, lai, model_data):
    height, width = ndvi.shape
    stacked = np.stack([ndvi.flatten(), savi.flatten(), lai.flatten()], axis=1)
    valid_mask = ~np.any(np.isnan(stacked), axis=1) & ~np.any(np.isinf(stacked), axis=1)
    
    pc1_flat = np.full(height * width, np.nan)
    pc2_flat = np.full(height * width, np.nan)
    
    if np.sum(valid_mask) > 0:
        valid_data = stacked[valid_mask]
        scaled_data = model_data['scaler'].transform(valid_data)
        pc_scores = model_data['pca'].transform(scaled_data)
        pc1_flat[valid_mask] = pc_scores[:, 0]
        pc2_flat[valid_mask] = pc_scores[:, 1]
    
    return pc1_flat.reshape(height, width), pc2_flat.reshape(height, width)

def save_raster(array, output_path, transform, crs, nodata=-9999):
    array_out = np.where(np.isnan(array), nodata, array)
    with rasterio.open(
        output_path, 'w', driver='GTiff',
        height=array.shape[0], width=array.shape[1], count=1,
        dtype=array_out.dtype, crs=crs, transform=transform,
        nodata=nodata, compress='lzw'
    ) as dst:
        dst.write(array_out, 1)

def get_available_dates():
    """Get all dates that have trained single-date PCA models."""
    model_files = glob.glob(
        os.path.join(BATCH_CONFIG['models_dir'], 'pca_*.pkl')
    )

    dates = []
    for f in model_files:
        basename = os.path.basename(f)

        # Skip multidate model explicitly
        if basename == 'pca_multidate.pkl':
            continue

        # Extract date
        date_str = basename.replace('pca_', '').replace('.pkl', '')

        # Basic validation
        try:
            datetime.strptime(date_str, '%Y-%m-%d')
            dates.append(date_str)
        except ValueError:
            continue

    return sorted(dates)


def process_single_date(study_area, target_date, model_data):
    """Process a single date and generate rasters."""
    print(f"\n{'='*60}")
    print(f"Processing: {target_date}")
    print(f"{'='*60}")
    
    # Get image
    image, actual_date = get_sentinel2_image(study_area, target_date)
    
    if image is None:
        print(f"  ❌ No image available for {target_date}")
        return False
    
    print(f"  ✓ Found image from {actual_date}")
    
    # Download data
    try:
        ndvi, savi, lai, transform, crs = image_to_array(image, study_area, BATCH_CONFIG['scale'])
        print(f"  ✓ Downloaded data: {ndvi.shape}")
    except Exception as e:
        print(f"  ❌ Download failed: {e}")
        return False
    
    # Apply PCA
    try:
        pc1, pc2 = apply_pca(ndvi, savi, lai, model_data)
        valid_pct = 100 * np.sum(~np.isnan(pc1)) / pc1.size
        print(f"  ✓ Applied PCA: {valid_pct:.1f}% valid pixels")
    except Exception as e:
        print(f"  ❌ PCA failed: {e}")
        return False
    
    # Save rasters
    try:
        date_str = actual_date.replace('-', '')
        pc1_path = os.path.join(BATCH_CONFIG['raster_dir'], f'PC1_{date_str}.tif')
        pc2_path = os.path.join(BATCH_CONFIG['raster_dir'], f'PC2_{date_str}.tif')
        
        save_raster(pc1, pc1_path, transform, crs)
        save_raster(pc2, pc2_path, transform, crs)
        print(f"  ✓ Saved: {os.path.basename(pc1_path)}, {os.path.basename(pc2_path)}")
        
        return True
    except Exception as e:
        print(f"  ❌ Save failed: {e}")
        return False

def create_time_series_animation():
    """Create animated GIF showing PC1 evolution over time."""
    print("\nCreating time-series animation...")
    
    pc1_files = sorted(glob.glob(os.path.join(BATCH_CONFIG['raster_dir'], 'PC1_*.tif')))
    
    if len(pc1_files) < 2:
        print("  Not enough files for animation")
        return
    
    fig, ax = plt.subplots(figsize=(10, 8))
    images = []
    
    for pc1_file in pc1_files:
        with rasterio.open(pc1_file) as src:
            data = src.read(1)
            data = np.where(data == -9999, np.nan, data)
            
            date_str = os.path.basename(pc1_file).replace('PC1_', '').replace('.tif', '')
            date_formatted = f"{date_str[:4]}-{date_str[4:6]}-{date_str[6:]}"
            
            ax.clear()
            valid_data = data[~np.isnan(data)]
            vmin, vmax = np.percentile(valid_data, [2, 98]) if len(valid_data) > 0 else (0, 1)
            
            im = ax.imshow(data, cmap='RdYlBu_r', vmin=vmin, vmax=vmax)
            ax.set_title(f'PC1 - {date_formatted}', fontsize=14, fontweight='bold')
            ax.axis('off')
            
            # Save frame
            frame_path = os.path.join(BATCH_CONFIG['raster_dir'], f'frame_{date_str}.png')
            plt.savefig(frame_path, dpi=150, bbox_inches='tight')
            images.append(frame_path)
    
    plt.close()
    
    # Create GIF (requires imageio)
    try:
        import imageio
        gif_path = os.path.join(BATCH_CONFIG['raster_dir'], 'PC1_timeseries.gif')
        
        with imageio.get_writer(gif_path, mode='I', duration=0.5) as writer:
            for frame_path in images:
                image = imageio.imread(frame_path)
                writer.append_data(image)
                os.remove(frame_path)  # Clean up
        
        print(f"  ✓ Saved animation: {gif_path}")
    except ImportError:
        print("  ℹ Install imageio to create GIF: pip install imageio")
